快速开始:
  1) scp v9-embedding-deploy-v1.2.0.tar.gz user@server:/opt/
  2) ssh user@server
  3) tar -xzf v9-embedding-deploy-v1.2.0.tar.gz -C /opt/finsight-v9/
  4) cd /opt/finsight-v9/backend && bash deploy.sh deploy --workers 1

验证命令:
  ./deploy.sh test --port 8001
  curl http://localhost:8001/api/embed/health
